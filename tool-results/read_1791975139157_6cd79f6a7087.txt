    73→export function washTick(st: DexState, tick: number, mIdx: number, base: string, quote: string, pushOp: PushOp): void {
    74→  const pk = poolKey(base, quote);
    75→  const p = getPool(st, pk);
    76→  if (!p || p.ra <= 0 || p.rb <= 0) return;
    77→  if (RAIL_OWNED_POOLS.has(pk)) return; // מסדרון-העוגן — זרימת-השטיפה לא נוגעת (R26)
    78→  if ((tick + mIdx) % WASH_ROTATE !== 0) return; // רוטציה — שתי-בריכות-מנוחות-לכל-בריכה-פועלת
    79→  // FIX(T25 · wash-anchor-rule, v2): הכיוון נגזר-מסטיית-הבריכה-מעוגן-האורקל-שלה —
    80→  // לא-מזוגיות-טיק ולא-מתיק-השטיפה (שני-הקודמים נמדדו-חי-בסימולציה והיו משאבות-
    81→  // מבניות: זוגיות-על-תיק-חד-צדדי דחפה בריכת-יציב 32× מהעוגן; תיק-משותף-חוצה-בריכות
    82→  // דחף-USDS-לכל-בריכה). השטיפה-היא עתה מנוע-נזילות-שומר-עוגן: בריכה-יקרה-מהעוגן
    83→  // מקבלת-מכירה-פנימה, זולה — קנייה, על-העוגן — שטיפה-מאוזנת-שני-הכיוונים. הכל
    84→  // ממלאי-אמת של-הצף (אין-צד-שאין-מלאי-עבורו — דילוג-כנה), אפס-רנדומיות.
    85→  const midA = poolMidFor(p, p.a, p.b); // µ-ציטוט ל-1 יחידת a
    86→  const refA = st.assets[p.a]?.refPrice ?? 0;
    87→  if (!(midA > 0)) return;
    88→  const devBps = refA > 0 ? Math.floor(((midA - refA) * 10_000) / refA) : 0;
    89→  const pushA = devBps > 30 ? true : devBps < -30 ? false : tick % 2 === 0;
    90→  const assetIn = (pushA ? p.a : p.b) as Asset;
    91→  const have = avail(st, WASH_WALLET, assetIn);
    92→  if (!(have > 0)) return;             // אין-מלאי-לצד — דילוג-כנה (אפס-משאבה-חד-צדדית)
    93→  const reserve = pushA ? p.ra : p.rb;
    94→  const want = Math.min(Math.floor((reserve * WASH_FRAC_NUM) / WASH_FRAC_DEN), WASH_CAP_PER_TRADE);
    95→  const amt = Math.min(want, Math.floor((have * 9) / 10));
    96→  if (amt > 0) {
    97→    pushOp({ type: "pool_swap", issuer: WASH_WALLET, nonce: `wash:a${tick}:${base}`, pair: pk, assetIn, amount: amt });
    98→  }
    99→}

   102→export function fairPrice(st: DexState, base: string, quote: string): number {
   103→  const ref = st.assets[base]?.refPrice ?? 0;
   104→  // R26 · עוגן-יחיד לזוגות-המסילה: הפייר של זוג-בבעלות-המסילה הוא refPrice — לא
   105→  // תערובת-ספר+בריכה. המצאו-החי (h18553): 80% מהפייר היה self-referential
   106→  // (ספר 50% + בריכה 30%) — נקודת-שבת-עצמית שהחזיקה-את-הספר ב-305µ בעוד-העוגן
   107→  // 54µ (STEEM-אמת). "מנוע-אחד-לעוגן" (SETTLE-16) — המסילה היא בעלת-המשמעות,
   108→  // והעוגן שלה הוא-האורקל, תמיד. הספר מצטט-סביב-העוגן ± spread — נקודה.
   109→  if (ref > 0 && RAIL_OWNED_POOLS.has(poolKey(base, quote))) return ref;
   110→  const bids = sortBook(st, base, quote, "buy");
   111→  const asks = sortBook(st, base, quote, "sell");
   112→  const bestBid = bids[0]?.price ?? 0;
   113→  const bestAsk = asks[0]?.price ?? 0;
   114→  const bookMid = bestBid > 0 && bestAsk > 0 ? Math.floor((bestBid + bestAsk) / 2) : (bestBid || bestAsk || 0);
   115→  const p = getPool(st, poolKey(base, quote));
   116→  const pm = p ? poolMidFor(p, base, quote) : 0;
   117→  // משקלים: ספר 50% · בריכה 30% · סולם-ערך 20% (נפילה-חן למקורות-קיימים)
   118→  const parts: [number, number][] = [];
   119→  if (bookMid > 0) parts.push([bookMid, 0.5]);
   120→  if (pm > 0) parts.push([pm, 0.3]);
   121→  if (ref > 0) parts.push([ref, 0.2]);
   122→  if (parts.length === 0) return ref;
   123→  const wsum = parts.reduce((s, [, w]) => s + w, 0);
   124→  return Math.floor(parts.reduce((s, [v, w]) => s + v * w, 0) / wsum);
   125→}

   240→function submit(rt: Map<string, SoldierRT>, st: DexState | null, soldier: string, op: Omit<Op, "issuer" | "nonce">, pushOp: PushOp): void {
   241→  const r = rt.get(soldier) ?? { nonceSeq: st ? seedNonceSeq(st, soldier) : 0, fills: 0, takes: 0 };
   242→  // FIX(EXEC-CD-9 · skip-occupied): בדיקת-תפוסה O(1) מול-ה-Set-האמת לפני-הדחיפה —
   243→  // הסקוו מדלג-מעל-כל-נונס-שכבר-חי (שרידי-ריצות/rewinds) במקום-סערת-replay-rejected.
   244→  // אותו-פורמט-מפתח כמו-applyOp: `${issuer}:${nonce}`.
   245→  if (st) {
   246→    let n = r.nonceSeq + 1;
   247→    while (nonceExists(st, `${soldier}:${soldier}:a${n}`)) n++;
   248→    r.nonceSeq = n;
   249→  } else {
   250→    r.nonceSeq++;
   251→  }
   252→  rt.set(soldier, r);
   253→  // FIX(R45 · nonce-owner): ה-nonce היה `a${seq}` חסר-בעלים — ה-seeding למעלה סרק
   254→  // קידומת `${soldier}:a` שלעולם-לא-נכתבה (קוד-מת) ← כל-אתחול-תהליך מתחיל-seq-מ-1
   255→  // מחדש, ההזמנות של כל-החיילים-והבוטים מתנגשות-ביניהן ובחלון-הנונסים-ההיסטורי
   256→  // (replay-rejected המוני-עד-שה-seq-עובר-את-החלון — בדיוק-מה-שההערה-למעלה-התכוונה-
   257→  // למנוע). עכשיו: הקידומת-בפועל-שווה-לזו-שה-seeding-סורק — הזריעה-חיה-וההתנגשות-נעלמת.
   258→  pushOp({ ...op, issuer: soldier, nonce: `${soldier}:a${r.nonceSeq}` } as Op);
   259→}

   278→export function pegkeeperTick(st: DexState, tick: number, pushOp: PushOp): void {
   279→  const m = MARKETS[0]; // הזוג-הריבוני — לחגורת-הסולם
   280→  const fair = fairPrice(st, m.base, m.quote);
   281→  if (fair <= 0) return;
   282→
   283→  // (1) מנעול-בריכה — לכל-הזוגות (FIX SETTLE-14: היה רק-הזוג-הריבוני ובריכת
   284→  // WSTEEM צפה חופשי — נסחרה עד 6x מעל הספר בלי-שום-גרביטציה)
   285→  // R26 · הסרת-הפטור-של-זוגות-המסילה: הפטור נולד (SETTLE-16) כי-הפייר-הצף גרר-
   286→  // את-הבריכה-בחזרה-מעלה אחרי-ה-REPEG ("מלחמת-סוכנים"). מאז-R26 הפייר-של-זוג-
   287→  // מסילה ≡ העוגן (fairPrice מחזיר refPrice) — אין-עוד-שני-מנועים-עם-שני-יעדים:
   288→  // מנוע-אחד, יעד-אחד (העוגן). המנעול חוזר-לשמור-גם-על-בריכת-המסדרון — בלי-
   289→  // הכוח-הזה בריכה-רדודה (אחרי-סימון-מחדש) נסחפת-חופשית-ברעש-הבוטים (נמדד:
   290→  // 54→66µ ב-60-טיקים) והמסילה חסרת-מלאי-לשפשף (doses=0). המנוע-השני נשאר
   291→  // למנות-העדינות (TWAP-scrubbing) — שניהם-פונים-לאותו-עוגן.
   292→  for (const mk of MARKETS) {
   293→    const mkPair = poolKey(mk.base, mk.quote);
   294→    const p = getPool(st, mkPair);
   295→    if (!p) continue;
   296→    const pm = poolMidFor(p, mk.base, mk.quote);
   297→    if (pm <= 0) continue;
   298→    const mkFair = fairPrice(st, mk.base, mk.quote); // פייר-פר-זוג — לעולם לא פייר-זר
   299→    if (mkFair <= 0) continue;
   300→    const gapBps = Math.floor(((pm - mkFair) / mkFair) * 10_000);
   301→    // SETTLE-14: פער-גדול (≥200bps) מטופל כל-טיק; קטן — כל-טיק-שני (חיסכון-גז)
   302→    if (Math.abs(gapBps) >= 30 && (Math.abs(gapBps) >= 200 || tick % 2 === 0)) {
   303→      const assetIn = gapBps > 0 ? mk.base : mk.quote; // יקר-בבריכה → מזרימים-בסיס-פנימה (מוריד-mid)
   304→      // R26 · מנה-מדויקת-שורש (סגירת-פער-לאפס): הליניאריזציה-מחצית-הפער היתה
   305→      // כפולה במכפיל — mid ∝ ra/rb (שינוי-יחס), אבל המנה חושבה-ליניארית-על-המונה
   306→      // בלבד. על-בריכה-רדודה מנת-10kµ הרימה-את-ה-mid ב-+3.7% (דיוק-היעד: +1.9%)
   307→      // → הצד-השני לא-החזיר-אותה → ראצ'ט מתמשך שדילדל-את-הבריכה מ-WSTEEM
   308→      // (נמדד-חי: 54→82µ תוך-240-טיקים). המנה-המדויקת להביא-את-מחיר-הבריכה
   309→      // בדיוק-לפייר: sqrt — שכן mid' = 1000·rQ·rB/(rB+x)² למכירה,
   310→      // ו-mid' = 1000(rQ+x)²/(rB·rQ) לקנייה. לפערים-קטנים היא מתכנסת-לליניארית
   311→      // (sqrt(1+g)≈1+g/2) — שיפור-טהור, בלי-overshoot, בלי-ראצ'ט.
   312→      const rBase = mk.base === p.a ? p.ra : p.rb;
   313→      const rQuote = mk.quote === p.a ? p.ra : p.rb;
   314→      const raw = gapBps > 0
   315→        ? Math.floor(Math.sqrt(Math.floor((1000 * rQuote * rBase) / mkFair)) - rBase)   // מכירת-בסיס פנימה
   316→        : Math.floor(Math.sqrt(Math.floor((mkFair * rBase * rQuote) / 1000)) - rQuote); // קניית-בסיס מהבריכה
   317→      const size = Math.min(Math.max(raw, 1), 1_200_000, avail(st, "arena", assetIn)); // IGNITION-R1: תקרת-המנעול ×2 (היתה 600k) — יישור-בריכות מהיר-יותר = עמלות-אמת
   318→      if (size >= 500) {
   319→        pushOp({ type: "pool_swap", issuer: "arena", nonce: `pk-pool-h${st.height}-${mk.base}`, pair: poolKey(mk.base, mk.quote), assetIn, amount: size });
   320→      }
   321→    }
   322→  }
   323→
   324→  // (2) חגורת-סולם: bookMid ∈ ref±400bps
   325→  const ref = st.assets[m.base]?.refPrice ?? 0;
   326→  if (ref > 0 && tick % 4 === 0) {
   327→    const bb = sortBook(st, m.base, m.quote, "buy")[0]?.price ?? 0;
   328→    const ba = sortBook(st, m.base, m.quote, "sell")[0]?.price ?? 0;
   329→    const mid = bb > 0 && ba > 0 ? Math.floor((bb + ba) / 2) : (bb || ba);
   330→    if (mid > 0) {
   331→      const devBps = Math.floor(((mid - ref) / ref) * 10_000);
   332→      // SETTLE-14 price-truth: גודל-החגורה פרופורציונלי לסטייה — 80k-קבוע מול
   333→      // סטיית-15% היה טיפה-בים (הספר צף ולא חזר). 120k+300µ/סט-בפס, תקרה 600k.
   334→      const bandQty = Math.min(120_000 + Math.abs(devBps) * 300, 600_000);
   335→      if (devBps > 400) {
   336→        // יקר-מדי-מול-הסולם → הארנה מוכרת-לתוך-הבידים (מורידה-את-המגע)
   337→        const qty = Math.min(bandQty, avail(st, "arena", m.base));
   338→        if (qty >= 5_000) {
   339→          pushOp({ type: "fill", issuer: "arena", nonce: `pk-band-h${st.height}-${m.base}-s`, base: m.base, quote: m.quote, side: "sell", amount: qty, market: true });
   340→        }
   341→      } else if (devBps < -400) {
   342→        // זול-מדי → הארנה קונה-מהאסקים (מרימה-את-המגע)
   343→        const px = ba > 0 ? ba : ref;
   344→        const qty = Math.min(bandQty, Math.floor((avail(st, "arena", m.quote) * 900) / (px * 10)));
   345→        if (qty >= 5_000) {
   346→          pushOp({ type: "fill", issuer: "arena", nonce: `pk-band-h${st.height}-${m.base}-b`, base: m.base, quote: m.quote, side: "buy", amount: qty, market: true });
   347→        }
   348→      }
   349→    }
   350→  }
   351→
   352→  // (3) רצפות-עומק (SETTLE-14): "הבריכות יהיו זמינות" — בריכה-שרזרבה-אחת-שלה
   353→  // מתחת-לרצפה מקבלת-השלמה-פרופורציונלית-מהארנה (הוספה-ביחס-הנוכחי = אפס-הזזת-mid).
   354→  // ממומן-רק-ממלאי-אמת-של-הארנה (בדיקת-avail לפני-הדחיפה — אפס-דחיות-ספר).
   355→  if (tick % 8 === 0) {
   356→    for (const m of MARKETS) {
   357→      const p = getPool(st, poolKey(m.base, m.quote));
   358→      if (!p) continue;
   359→      const floorA = POOL_DEPTH_FLOORS[p.a] ?? 0;
   360→      const floorB = POOL_DEPTH_FLOORS[p.b] ?? 0;
   361→      const shortA = Math.max(0, floorA - p.ra);
   362→      const shortB = Math.max(0, floorB - p.rb);
   363→      if (shortA <= 0 && shortB <= 0) continue;
   364→      // משלימים-את-הצד-החסר-הגדול; הצד-השני נגזר-מיחס-הבריכה (חוזה pool_add)
   365→      const addA = shortA >= shortB ? shortA : (p.ra > 0 ? Math.floor((shortB * p.ra) / p.rb) : 0);
   366→      if (p.ra <= 0 || p.rb <= 0 || addA <= 0) continue;
   367→      const addB = Math.floor((p.rb * addA) / p.ra);
   368→      if (addA < 20_000 || addB < 20_000) continue;
   369→      if (avail(st, "arena", p.a) < addA || avail(st, "arena", p.b) < addB) continue;
   370→      pushOp({ type: "pool_add", issuer: "arena", nonce: `pk-floor-h${st.height}-${p.a}`, pair: poolKey(p.a, p.b), amount: addA });
   371→    }
   372→  }
