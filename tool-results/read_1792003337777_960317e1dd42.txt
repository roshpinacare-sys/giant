    83→function parseMeta(meta: string | null): { llmError?: boolean; sources?: string[] } {
    84→  if (!meta) return {};
    85→  try {
    86→    const parsed: unknown = JSON.parse(meta);
    87→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
    88→      const rec = parsed as Record<string, unknown>;
    89→      return {
    90→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
    91→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
    92→      };
    93→    }
    94→    return {};
    95→  } catch {
    96→    return {};
    97→  }
    98→}

   100→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   101→function computeSummary(messages: SovereignMessage[]): Summary {
   102→  const s: Summary = {
   103→    unreadForOwner: 0,
   104→    pendingRequests: 0,
   105→    pendingAmount: 0,
   106→    pendingCurrency: null,
   107→    relayUnprocessed: 0,
   108→  };
   109→  for (const m of messages) {
   110→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   111→    if (m.requestStatus === "PENDING") {
   112→      s.pendingRequests += 1;
   113→      if (typeof m.amount === "number") {
   114→        s.pendingAmount += m.amount;
   115→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   116→      }
   117→    }
   118→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   119→  }
   120→  return s;
   121→}

   123→function mergeMessages(
   124→  prev: SovereignMessage[],
   125→  incoming: SovereignMessage[],
   126→): SovereignMessage[] {
   127→  const byId = new Map<string, SovereignMessage>();
   128→  for (const m of prev) byId.set(m.id, m);
   129→  for (const m of incoming) byId.set(m.id, m);
   130→  return Array.from(byId.values()).sort(
   131→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   132→  );
   133→}
