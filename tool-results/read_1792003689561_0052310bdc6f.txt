    83→    83→function parseMeta(meta: string | null): { llmError?: boolean; sources?: string[] } {
    84→    84→  if (!meta) return {};
    85→    85→  try {
    86→    86→    const parsed: unknown = JSON.parse(meta);
    87→    87→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
    88→    88→      const rec = parsed as Record<string, unknown>;
    89→    89→      return {
    90→    90→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
    91→    91→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
    92→    92→      };
    93→    93→    }
    94→    94→    return {};
    95→    95→  } catch {
    96→    96→    return {};
    97→    97→  }
    98→    98→}

   101→   101→function computeSummary(messages: SovereignMessage[]): Summary {
   102→   102→  const s: Summary = {
   103→   103→    unreadForOwner: 0,
   104→   104→    pendingRequests: 0,
   105→   105→    pendingAmount: 0,
   106→   106→    pendingCurrency: null,
   107→   107→    relayUnprocessed: 0,
   108→   108→  };
   109→   109→  for (const m of messages) {
   110→   110→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   111→   111→    if (m.requestStatus === "PENDING") {
   112→   112→      s.pendingRequests += 1;
   113→   113→      if (typeof m.amount === "number") {
   114→   114→        s.pendingAmount += m.amount;
   115→   115→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   116→   116→      }
   117→   117→    }
   118→   118→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   119→   119→  }
   120→   120→  return s;
   121→   121→}

   123→   123→function mergeMessages(
   124→   124→  prev: SovereignMessage[],
   125→   125→  incoming: SovereignMessage[],
   126→   126→): SovereignMessage[] {
   127→   127→  const byId = new Map<string, SovereignMessage>();
   128→   128→  for (const m of prev) byId.set(m.id, m);
   129→   129→  for (const m of incoming) byId.set(m.id, m);
   130→   130→  return Array.from(byId.values()).sort(
   131→   131→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   132→   132→  );
   133→   133→}
