   152→function parseMeta(meta: string | null): {
   153→  llmError?: boolean;
   154→  sources?: string[];
   155→  gitLand?: GitLandInfo;
   156→} {
   157→  if (!meta) return {};
   158→  try {
   159→    const parsed: unknown = JSON.parse(meta);
   160→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   161→      const rec = parsed as Record<string, unknown>;
   162→      return {
   163→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
   164→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
   165→        gitLand:
   166→          rec.gitLand && typeof rec.gitLand === "object" && !Array.isArray(rec.gitLand)
   167→            ? (rec.gitLand as GitLandInfo)
   168→            : undefined,
   169→      };
   170→    }
   171→    return {};
   172→  } catch {
   173→    return {};
   174→  }
   175→}

   177→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   178→function computeSummary(messages: SovereignMessage[]): Summary {
   179→  const s: Summary = {
   180→    unreadForOwner: 0,
   181→    pendingRequests: 0,
   182→    pendingAmount: 0,
   183→    pendingCurrency: null,
   184→    relayUnprocessed: 0,
   185→  };
   186→  for (const m of messages) {
   187→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   188→    if (m.requestStatus === "PENDING") {
   189→      s.pendingRequests += 1;
   190→      if (typeof m.amount === "number") {
   191→        s.pendingAmount += m.amount;
   192→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   193→      }
   194→    }
   195→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   196→  }
   197→  return s;
   198→}

   200→function mergeMessages(
   201→  prev: SovereignMessage[],
   202→  incoming: SovereignMessage[],
   203→): SovereignMessage[] {
   204→  const byId = new Map<string, SovereignMessage>();
   205→  for (const m of prev) byId.set(m.id, m);
   206→  for (const m of incoming) byId.set(m.id, m);
   207→  return Array.from(byId.values()).sort(
   208→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   209→  );
   210→}
