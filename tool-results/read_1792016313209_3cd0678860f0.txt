   153→function parseMeta(meta: string | null): {
   154→  llmError?: boolean;
   155→  sources?: string[];
   156→  gitLand?: GitLandInfo;
   157→} {
   158→  if (!meta) return {};
   159→  try {
   160→    const parsed: unknown = JSON.parse(meta);
   161→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   162→      const rec = parsed as Record<string, unknown>;
   163→      return {
   164→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
   165→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
   166→        gitLand:
   167→          rec.gitLand && typeof rec.gitLand === "object" && !Array.isArray(rec.gitLand)
   168→            ? (rec.gitLand as GitLandInfo)
   169→            : undefined,
   170→      };
   171→    }
   172→    return {};
   173→  } catch {
   174→    return {};
   175→  }
   176→}

   178→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   179→function computeSummary(messages: SovereignMessage[]): Summary {
   180→  const s: Summary = {
   181→    unreadForOwner: 0,
   182→    pendingRequests: 0,
   183→    pendingAmount: 0,
   184→    pendingCurrency: null,
   185→    relayUnprocessed: 0,
   186→  };
   187→  for (const m of messages) {
   188→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   189→    if (m.requestStatus === "PENDING") {
   190→      s.pendingRequests += 1;
   191→      if (typeof m.amount === "number") {
   192→        s.pendingAmount += m.amount;
   193→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   194→      }
   195→    }
   196→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   197→  }
   198→  return s;
   199→}

   201→function mergeMessages(
   202→  prev: SovereignMessage[],
   203→  incoming: SovereignMessage[],
   204→): SovereignMessage[] {
   205→  const byId = new Map<string, SovereignMessage>();
   206→  for (const m of prev) byId.set(m.id, m);
   207→  for (const m of incoming) byId.set(m.id, m);
   208→  return Array.from(byId.values()).sort(
   209→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   210→  );
   211→}
