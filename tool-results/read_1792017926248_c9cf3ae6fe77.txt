   155→function parseMeta(meta: string | null): {
   156→  llmError?: boolean;
   157→  sources?: string[];
   158→  gitLand?: GitLandInfo;
   159→} {
   160→  if (!meta) return {};
   161→  try {
   162→    const parsed: unknown = JSON.parse(meta);
   163→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   164→      const rec = parsed as Record<string, unknown>;
   165→      return {
   166→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
   167→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
   168→        gitLand:
   169→          rec.gitLand && typeof rec.gitLand === "object" && !Array.isArray(rec.gitLand)
   170→            ? (rec.gitLand as GitLandInfo)
   171→            : undefined,
   172→      };
   173→    }
   174→    return {};
   175→  } catch {
   176→    return {};
   177→  }
   178→}

   180→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   181→function computeSummary(messages: SovereignMessage[]): Summary {
   182→  const s: Summary = {
   183→    unreadForOwner: 0,
   184→    pendingRequests: 0,
   185→    pendingAmount: 0,
   186→    pendingCurrency: null,
   187→    relayUnprocessed: 0,
   188→  };
   189→  for (const m of messages) {
   190→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   191→    if (m.requestStatus === "PENDING") {
   192→      s.pendingRequests += 1;
   193→      if (typeof m.amount === "number") {
   194→        s.pendingAmount += m.amount;
   195→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   196→      }
   197→    }
   198→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   199→  }
   200→  return s;
   201→}

   203→function mergeMessages(
   204→  prev: SovereignMessage[],
   205→  incoming: SovereignMessage[],
   206→): SovereignMessage[] {
   207→  const byId = new Map<string, SovereignMessage>();
   208→  for (const m of prev) byId.set(m.id, m);
   209→  for (const m of incoming) byId.set(m.id, m);
   210→  return Array.from(byId.values()).sort(
   211→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   212→  );
   213→}
