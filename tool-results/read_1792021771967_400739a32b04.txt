   157→function parseMeta(meta: string | null): {
   158→  llmError?: boolean;
   159→  sources?: string[];
   160→  gitLand?: GitLandInfo;
   161→} {
   162→  if (!meta) return {};
   163→  try {
   164→    const parsed: unknown = JSON.parse(meta);
   165→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   166→      const rec = parsed as Record<string, unknown>;
   167→      return {
   168→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
   169→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
   170→        gitLand:
   171→          rec.gitLand && typeof rec.gitLand === "object" && !Array.isArray(rec.gitLand)
   172→            ? (rec.gitLand as GitLandInfo)
   173→            : undefined,
   174→      };
   175→    }
   176→    return {};
   177→  } catch {
   178→    return {};
   179→  }
   180→}

   182→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   183→function computeSummary(messages: SovereignMessage[]): Summary {
   184→  const s: Summary = {
   185→    unreadForOwner: 0,
   186→    pendingRequests: 0,
   187→    pendingAmount: 0,
   188→    pendingCurrency: null,
   189→    relayUnprocessed: 0,
   190→  };
   191→  for (const m of messages) {
   192→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   193→    if (m.requestStatus === "PENDING") {
   194→      s.pendingRequests += 1;
   195→      if (typeof m.amount === "number") {
   196→        s.pendingAmount += m.amount;
   197→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   198→      }
   199→    }
   200→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   201→  }
   202→  return s;
   203→}

   205→function mergeMessages(
   206→  prev: SovereignMessage[],
   207→  incoming: SovereignMessage[],
   208→): SovereignMessage[] {
   209→  const byId = new Map<string, SovereignMessage>();
   210→  for (const m of prev) byId.set(m.id, m);
   211→  for (const m of incoming) byId.set(m.id, m);
   212→  return Array.from(byId.values()).sort(
   213→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   214→  );
   215→}
