   154→function parseMeta(meta: string | null): {
   155→  llmError?: boolean;
   156→  sources?: string[];
   157→  gitLand?: GitLandInfo;
   158→} {
   159→  if (!meta) return {};
   160→  try {
   161→    const parsed: unknown = JSON.parse(meta);
   162→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   163→      const rec = parsed as Record<string, unknown>;
   164→      return {
   165→        llmError: typeof rec.llmError === "boolean" ? rec.llmError : undefined,
   166→        sources: Array.isArray(rec.sources) ? rec.sources.map(String) : undefined,
   167→        gitLand:
   168→          rec.gitLand && typeof rec.gitLand === "object" && !Array.isArray(rec.gitLand)
   169→            ? (rec.gitLand as GitLandInfo)
   170→            : undefined,
   171→      };
   172→    }
   173→    return {};
   174→  } catch {
   175→    return {};
   176→  }
   177→}

   179→// מראה את חוקי-הסיכום של השרת — כדי-שהסטטיסטיקה-בלקוח תמיד תתלכד עם ה-API
   180→function computeSummary(messages: SovereignMessage[]): Summary {
   181→  const s: Summary = {
   182→    unreadForOwner: 0,
   183→    pendingRequests: 0,
   184→    pendingAmount: 0,
   185→    pendingCurrency: null,
   186→    relayUnprocessed: 0,
   187→  };
   188→  for (const m of messages) {
   189→    if (m.from === "SOVEREIGNTY" && m.status === "UNREAD") s.unreadForOwner += 1;
   190→    if (m.requestStatus === "PENDING") {
   191→      s.pendingRequests += 1;
   192→      if (typeof m.amount === "number") {
   193→        s.pendingAmount += m.amount;
   194→        if (!s.pendingCurrency && m.currency) s.pendingCurrency = m.currency;
   195→      }
   196→    }
   197→    if (m.from === "AGENT_RELAY" && !m.relayProcessed) s.relayUnprocessed += 1;
   198→  }
   199→  return s;
   200→}

   202→function mergeMessages(
   203→  prev: SovereignMessage[],
   204→  incoming: SovereignMessage[],
   205→): SovereignMessage[] {
   206→  const byId = new Map<string, SovereignMessage>();
   207→  for (const m of prev) byId.set(m.id, m);
   208→  for (const m of incoming) byId.set(m.id, m);
   209→  return Array.from(byId.values()).sort(
   210→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   211→  );
   212→}
