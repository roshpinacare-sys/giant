    32→const heNum = new Intl.NumberFormat("he-IL");
    33→
    34→function fmtNum(n: number): string {
    35→  if (!Number.isFinite(n)) return "—";
    36→  return heNum.format(Math.round(n * 100) / 100);
    37→}
    38→
    39→function fmtBytes(bytes: number): string {
    40→  if (bytes < 1024) return `${bytes}B`;
    41→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)}KB`;
    42→  return `${(bytes / 1024 / 1024).toFixed(1)}MB`;
    43→}
    44→
    45→function fmtTime(iso: string | null): string {
    46→  if (!iso) return "—";
    47→  const d = new Date(iso);
    48→  if (Number.isNaN(d.getTime())) return "—";
    49→  return new Intl.DateTimeFormat("he-IL", {
    50→    day: "2-digit",
    51→    month: "2-digit",
    52→    hour: "2-digit",
    53→    minute: "2-digit",
    54→  }).format(d);
    55→}

    57→/* count-up (60fps.design לקח: המספרים-האמיתיים-הם-העיצוב — והם-נכנסים-חיים) */
    58→function useCountUp(target: number, durationMs = 900): number {
    59→  const [val, setVal] = useState(() => (Number.isFinite(target) ? 0 : target));
    60→  useEffect(() => {
    61→    if (!Number.isFinite(target)) return; // ערך-לא-מספרי: המסך-יציג "—" דרך-fmtNum
    62→    let raf = 0;
    63→    const t0 = performance.now();
    64→    const step = (t: number) => {
    65→      const p = Math.min(1, (t - t0) / durationMs);
    66→      const eased = 1 - Math.pow(1 - p, 3);
    67→      setVal(target * eased);
    68→      if (p < 1) raf = requestAnimationFrame(step);
    69→    };
    70→    raf = requestAnimationFrame(step);
    71→    return () => cancelAnimationFrame(raf);
    72→  }, [target, durationMs]);
    73→  return val;
    74→}
    75→
    76→function AnimatedNumber({ value, suffix }: { value: number; suffix?: string }) {
    77→  const v = useCountUp(value);
    78→  return (
    79→    <span className="tabular-nums">
    80→      {fmtNum(v)}
    81→      {suffix ? (
    82→        <span className="ms-1 text-[0.6em] font-semibold text-zinc-400">{suffix}</span>
    83→      ) : null}
    84→    </span>
    85→  );
    86→}
